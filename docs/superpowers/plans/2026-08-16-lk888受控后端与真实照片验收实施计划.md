```python
@dataclass(frozen=True)
class BackendConfig:
    lk888_api_key: str
    backend_token: str
    lk888_base_url: str = "https://api.lk888.ai"
    analysis_model: str = "gpt-4o"
    image_model: str = "gpt-image-2"
    host: str = "127.0.0.1"
    port: int = 8787
    state_dir: Path = Path("output/photo-avatar-backend")

    @classmethod
    def from_env(cls, env: Mapping[str, str]) -> "BackendConfig":
        def required(name: str) -> str:
            value = env.get(name, "").strip()
            if not value:
                raise ConfigError(f"missing {name}")
            return value

        base_url = env.get("LK888_BASE_URL", "https://api.lk888.ai").rstrip("/")
        parsed = urlparse(base_url)
        if parsed.scheme != "https" or not parsed.hostname:
            raise ConfigError("LK888_BASE_URL must use https")
        host = env.get("PHOTO_AVATAR_BACKEND_HOST", "127.0.0.1").strip()
        if host not in {"127.0.0.1", "localhost"}:
            raise ConfigError("development host must be loopback")
        state_dir = Path(
            env.get("PHOTO_AVATAR_BACKEND_STATE_DIR", "output/photo-avatar-backend")
        ).expanduser().resolve()
        return cls(
            lk888_api_key=required("LK888_API_KEY"),
            backend_token=required("PHOTO_AVATAR_BACKEND_TOKEN"),
            lk888_base_url=base_url,
            analysis_model=env.get("LK888_ANALYSIS_MODEL", "gpt-4o").strip(),
            image_model=env.get("LK888_IMAGE_MODEL", "gpt-image-2").strip(),
            host=host,
            port=int(env.get("PHOTO_AVATAR_BACKEND_PORT", "8787")),
            state_dir=state_dir,
        )
```

```python
class Lk888Client:
    def __init__(self, config: BackendConfig, http: httpx.Client):
        self.config = config
        self.http = http

    def analyze_json(self, prompt: str, images: Sequence[bytes], schema: dict) -> dict:
        content = [{"type": "text", "text": prompt}]
        content.extend({
            "type": "image_url",
            "image_url": {"url": "data:image/png;base64," + b64encode(image).decode("ascii")},
        } for image in images)
        response = self.http.post(
            f"{self.config.lk888_base_url}/v1/chat/completions",
            headers=self._headers(),
            json={
                "model": self.config.analysis_model,
                "messages": [{"role": "user", "content": content}],
                "response_format": {
                    "type": "json_schema",
                    "json_schema": {"name": "photo_avatar", "strict": True, "schema": schema},
                },
            },
        )
        self._require_success(response)
        choices = response.json().get("choices", [])
        if len(choices) != 1:
            raise Lk888Error("temporaryUnavailable", True, "invalid choices")
        try:
            return json.loads(choices[0]["message"]["content"])
        except (KeyError, TypeError, json.JSONDecodeError) as exc:
            raise Lk888Error("temporaryUnavailable", True, "invalid JSON content") from exc

    def submit_image(self, prompt: str, images: Sequence[bytes]) -> str:
        response = self.http.post(
            f"{self.config.lk888_base_url}/v1/media/generate",
            headers=self._headers(),
            json={
                "model": self.config.image_model,
                "prompt": prompt,
                "images": [b64encode(image).decode("ascii") for image in images],
                "n": 1,
                "size": "auto",
                "quality": "auto",
                "response_format": "url",
            },
        )
        self._require_success(response)
        task_id = response.json().get("data", {}).get("task_id")
        if not isinstance(task_id, str) or not task_id:
            raise Lk888Error("temporaryUnavailable", True, "missing task_id")
        return task_id

    def poll_image(self, task_id: str) -> MediaState:
        response = self.http.get(
            f"{self.config.lk888_base_url}/v1/media/status",
            headers=self._headers(),
            params={"task_id": task_id},
        )
        self._require_success(response)
        return MediaState.parse(response.json(), expected_task_id=task_id)

    def download(self, url: str) -> bytes:
        response = self.http.get(url, follow_redirects=False)
        self._require_success(response)
        if response.headers.get("content-type", "").split(";", 1)[0] != "image/png":
            raise Lk888Error("invalidInput", False, "artifact is not PNG")
        body = response.content
        if len(body) > 20 * 1024 * 1024:
            raise Lk888Error("invalidInput", False, "artifact exceeds 20 MiB")
        return body

    def _headers(self) -> dict[str, str]:
        return {"Authorization": f"Bearer {self.config.lk888_api_key}"}

    @staticmethod
    def _require_success(response: httpx.Response) -> None:
        if response.is_success:
            return
        if response.status_code in {401, 403}:
            raise Lk888Error("auth", False, "provider authentication failed")
        if response.status_code == 429:
            raise Lk888Error("temporaryUnavailable", True, "provider rate limited")
        if response.status_code >= 500:
            raise Lk888Error("provider5xx", True, "provider unavailable")
        raise Lk888Error("invalidInput", False, "provider rejected request")
```

文本请求使用 data URL 和严格 JSON response format；媒体请求保持 `n=1`、`size=auto`、`quality=auto`、`response_format=url`。客户端自身不重试提交，重试次数只由 Rust manager 管理。

```python
def create_app(config: BackendConfig, store: JobStore) -> FastAPI:
    app = FastAPI()

    def authorize(authorization: str = Header(default="")) -> None:
        expected = f"Bearer {config.backend_token}"
        if not secrets.compare_digest(authorization, expected):
            raise HTTPException(status_code=401, detail="unauthorized")

    @app.get("/healthz")
    def health() -> dict[str, str]:
        return {"status": "ok"}

    @app.post("/v1/photo-avatar/steps", dependencies=[Depends(authorize)])
    def submit(payload: dict) -> dict:
        job = store.submit(StepRequest.parse(payload))
        return {"providerSessionId": job.provider_session_id, "jobId": job.job_id}

    @app.get("/v1/photo-avatar/jobs/{job_id}", dependencies=[Depends(authorize)])
    def status(job_id: str) -> dict:
        return store.status(job_id).to_wire()

    @app.post("/v1/photo-avatar/jobs/{job_id}/cancel", dependencies=[Depends(authorize)])
    def cancel(job_id: str) -> dict:
        return store.cancel(job_id).to_wire()

    @app.delete("/v1/photo-avatar/sessions/{session_id}", dependencies=[Depends(authorize)])
    def delete_session(session_id: str) -> dict:
        return store.delete_session(session_id).to_wire()

    @app.get("/v1/photo-avatar/artifacts/{artifact_id}", dependencies=[Depends(authorize)])
    def artifact(artifact_id: str) -> Response:
        return Response(store.read_artifact(artifact_id), media_type="image/png")

    return app
```
