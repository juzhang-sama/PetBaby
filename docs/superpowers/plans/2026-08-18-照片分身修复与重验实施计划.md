- [ ] **Step 3: 实现最小合同与模板导出**

实现以下完整边界。`ModuleSemanticSnapshot` 由 `uv_guides.py` 从固定 body module 合同、neutral alpha 和七张 `L` 模式 mask 构造；`semantic_layers.py` 不自行猜测区域：

```python
import hashlib
from io import BytesIO

from PIL import Image

from .contracts import ContractError


def decode_png_exact(png: bytes, *, mode: str, size: tuple[int, int]) -> Image.Image:
    try:
        with Image.open(BytesIO(png)) as image:
            if image.format != "PNG":
                raise ContractError("semantic layer input must be a PNG")
            image.load()
            if image.mode != mode:
                raise ContractError(f"semantic layer input must use {mode} mode")
            if image.size != size:
                raise ContractError("semantic layer dimensions do not match the module")
            return image.copy()
    except ContractError:
        raise
    except (OSError, SyntaxError, ValueError) as exc:
        raise ContractError("semantic layer input must be a valid PNG") from exc


def build_semantic_layer_specs(
    module_snapshot: ModuleSemanticSnapshot,
) -> Sequence[SemanticLayerSpec]:
    expected = set(SEMANTIC_LAYER_IDS)
    if set(module_snapshot.layer_masks) != expected:
        raise ContractError("semantic layer masks do not match the fixed layer set")
    if set(module_snapshot.layer_mask_sha256) != expected:
        raise ContractError("semantic layer mask hashes do not match the fixed layer set")
    if set(module_snapshot.layer_anchors) != expected:
        raise ContractError("semantic layer anchors do not match the fixed layer set")
    if hashlib.sha256(module_snapshot.source_alpha).hexdigest() != module_snapshot.source_alpha_sha256:
        raise ContractError("module source alpha hash does not match")

    specs: list[SemanticLayerSpec] = []
    for layer_id in SEMANTIC_LAYER_IDS:
        anchor_x, anchor_y = module_snapshot.layer_anchors[layer_id]
        mask_png = module_snapshot.layer_masks[layer_id]
        mask_sha256 = hashlib.sha256(mask_png).hexdigest()
        if mask_sha256 != module_snapshot.layer_mask_sha256[layer_id]:
            raise ContractError(f"semantic layer mask hash does not match: {layer_id}")
        if not 0 <= anchor_x < module_snapshot.width or not 0 <= anchor_y < module_snapshot.height:
            raise ContractError(f"semantic layer anchor is outside the atlas: {layer_id}")
        specs.append(SemanticLayerSpec(
            layer_id=layer_id,
            width=module_snapshot.width,
            height=module_snapshot.height,
            anchor_x=anchor_x,
            anchor_y=anchor_y,
            mask_png=mask_png,
            mask_sha256=mask_sha256,
            seam_dilation_px=module_snapshot.seam_dilation_px,
        ))
    return tuple(specs)

def validate_semantic_layer_png(
    png: bytes,
    spec: SemanticLayerSpec,
) -> ValidatedSemanticLayer:
    layer = decode_png_exact(png, mode="RGBA", size=(spec.width, spec.height))
    mask = decode_png_exact(spec.mask_png, mode="L", size=(spec.width, spec.height))
    rgba = layer.tobytes()
    alpha = layer.getchannel("A").tobytes()
    for index, mask_value in enumerate(mask.tobytes()):
        offset = index * 4
        if mask_value == 0 and alpha[index] != 0:
            raise ContractError(f"semantic layer escapes its mask: {spec.layer_id}")
        if alpha[index] == 0 and rgba[offset:offset + 3] != b"\x00\x00\x00":
            raise ContractError(f"semantic layer has RGB under transparent alpha: {spec.layer_id}")
    output = BytesIO()
    layer.save(output, format="PNG", optimize=False, compress_level=9)
    canonical_png = output.getvalue()
    return ValidatedSemanticLayer(
        layer_id=spec.layer_id,
        canonical_png=canonical_png,
        rgba=rgba,
        provider_raw_sha256=hashlib.sha256(png).hexdigest(),
        canonical_layer_sha256=hashlib.sha256(canonical_png).hexdigest(),
        mask_sha256=spec.mask_sha256,
        anchor_x=spec.anchor_x,
        anchor_y=spec.anchor_y,
    )
```

`decode_png_exact` 复用 `texture_compositor.py` 的 PNG 解码错误语义并返回已加载的 Pillow `Image`；它验证 PNG、精确模式和精确尺寸。上述边界只验证格式、遮罩、透明 RGB、锚点和 layer ID，不做身份相似度判断。

- [ ] **Step 3: 实现逐层遮罩、扩边和固定顺序合成**

合成顺序固定为：`body-base`、`occlusion-underlay`、`chest-forelegs`、`tail`、`face`、`ears`、`eyes-eyelids`。每层先应用权威 mask，再在接缝安全区执行固定像素扩边；越界像素直接拒绝，不做自动缩放。

`texture_compositor.py` 对外固定为以下接口；`compose_semantic_atlas` 必须验证七个 layer ID 各出现一次、逐层 `mask_sha256` 与 `module_snapshot` 一致，并用 `module_snapshot.source_alpha` 写入最终 alpha：

```python
EXPECTED_LAYER_ORDER = (
    "body-base",
    "occlusion-underlay",
    "chest-forelegs",
    "tail",
    "face",
    "ears",
    "eyes-eyelids",
)

@dataclass(frozen=True)
class SemanticAtlas:
    png: bytes
    canonical_sha256: str
    source_alpha_sha256: str
    transparent_rgb_is_zero: bool
    layer_order: Sequence[str]

def compose_semantic_atlas(
    *,
    layers: Sequence[ValidatedSemanticLayer],
    module_snapshot: ModuleSemanticSnapshot,
) -> SemanticAtlas:
    """按 EXPECTED_LAYER_ORDER 合成唯一 canonical atlas。"""
```

实现体按固定顺序执行四步：先用对应 `L` mask 清除层外像素；再仅在该 mask 的 `seam_dilation_px` 安全区内做最近邻 RGB 扩边；随后按 `EXPECTED_LAYER_ORDER` 覆盖 RGB；最后逐像素写入 `source_alpha` 并将 alpha 为 0 的 RGB 清零。任何尺寸、hash、layer 集合或 mask 边界不匹配都抛出 `ContractError`，不得缩放、补层或改用旧整体 provider RGB。

- [ ] **Step 4: 接入逐层生成和独立重试**

`pipelines.py` 先锁定 `identity_reference_sha256`、`profile_sha256` 和 `ModuleSemanticSnapshot`，再按 `SEMANTIC_LAYER_IDS` 顺序调用现有 `client.submit_image` / `client.poll_image`。每次请求只携带同一身份参考图、同一 profile 摘要、当前 layer 模板和 mask；成功层立即冻结，不随其他层重试：

```python
validated_layers: dict[str, ValidatedSemanticLayer] = {}
for spec in build_semantic_layer_specs(module_snapshot):
    for attempt in range(1, 4):
        try:
            provider_png = render_one_semantic_layer(
                client=client,
                identity_reference=identity_reference,
                profile=profile,
                spec=spec,
                attempt=attempt,
            )
            validated = validate_semantic_layer_png(provider_png, spec)
        except (ContractError, Lk888Error) as exc:
            record_layer_attempt(spec.layer_id, attempt, exc)
            if isinstance(exc, Lk888Error) and not exc.retryable:
                raise
            if attempt == 3:
                raise
            continue
        validated_layers[spec.layer_id] = validated
        record_layer_success(spec.layer_id, attempt, validated)
        break
```

`render_one_semantic_layer` 复用当前 `render_texture_atlas` 的任务 ID 校验、轮询终态、超时和 20 MiB 下载上限；`record_layer_attempt`/`record_layer_success` 写入 Task 3 Step 5 的审计结构。测试必须证明：face 第二次成功时其余六层只提交一次；不可重试错误只提交一次；第三次仍失败时不调用 `compose_semantic_atlas`，也不产生可安装 artifact。
